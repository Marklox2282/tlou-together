cmake_minimum_required(VERSION 3.25)
project(tlou1-coop LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")

include(FetchContent)
set(SAFETYHOOK_FETCH_ZYDIS ON)
FetchContent_Declare(safetyhook
    GIT_REPOSITORY https://github.com/cursey/safetyhook
    GIT_TAG        629558c64009a7291ba6ed5cfb49187086a27a47)   # тот же пин, что у TLOU2-Coop
FetchContent_MakeAvailable(safetyhook)

get_filename_component(LLVM_BIN "${CMAKE_CXX_COMPILER}" DIRECTORY)
find_program(CMAKE_ASM_MASM_COMPILER NAMES llvm-ml64 ml64 HINTS "${LLVM_BIN}" REQUIRED)
enable_language(ASM_MASM)
add_library(winmm-thunks OBJECT src/winmm_thunks.asm)
add_library(tlou1-coop SHARED src/main.cpp src/engine.cpp src/puppet.cpp src/enemies.cpp src/menu.cpp src/log.cpp src/net.cpp src/update.cpp src/signing.cpp src/winmm-proxy.cpp src/winmm_exports.def)
target_compile_options(tlou1-coop PRIVATE /W4 /utf-8)
target_link_libraries(tlou1-coop PRIVATE winmm-thunks safetyhook ws2_32 winhttp bcrypt)
set_target_properties(tlou1-coop PROPERTIES OUTPUT_NAME winmm PREFIX "")
add_executable(smoke test/smoke.cpp)
add_executable(netsim test/netsim.cpp src/net.cpp)
target_compile_options(netsim PRIVATE /W4)
target_link_libraries(netsim PRIVATE ws2_32)
add_executable(enginetest test/enginetest.cpp src/engine.cpp)
add_executable(signer tools/signer.cpp src/signing.cpp)
target_link_libraries(signer PRIVATE bcrypt)
add_executable(updatetest test/updatetest.cpp src/update.cpp src/signing.cpp)
target_link_libraries(updatetest PRIVATE winhttp bcrypt)
